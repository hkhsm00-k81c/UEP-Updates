function recordRuleField(row,names){
  for(const name of names){
    const key=Object.keys(row||{}).find(k=>String(k).replace(/\s/g,"").includes(name));
    const value=key?String(row[key]??"").trim():"";
    if(value)return value;
  }

}
function applySchoolRecordGuardrails(text){
  let out=String(text||"").replace(/\s+/g," ").trim();
  // 33_생기부작성절대규칙과 35_금지표현DB는 학교 검증 규칙이므로 완화하지 않고 최종 강제 적용한다.
  recordRuleRowsBySource("35_금지표현DB").forEach(row=>{
    const forbidden=recordRuleField(row,["금지표현","금지어"]);
    if(!forbidden||forbidden.length>60)return;
    const enabled=recordRuleField(row,["사용여부","활성화여부"]);
    if(enabled&&/^N$/i.test(enabled))return;
    if(!out.includes(forbidden))return;
    const replacement=recordRuleField(row,["수정추천표현","추천표현","대체표현"]);
    const auto=recordRuleField(row,["자동치환여부","자동치환"]);
    if(replacement&&(!auto||/^Y$/i.test(auto)))out=out.split(forbidden).join(replacement);
  });
  return enforceTeacherEnding(out);
}
