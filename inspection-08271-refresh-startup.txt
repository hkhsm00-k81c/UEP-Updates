function applyProgramWritePatch08172(target,patch){
  if(!target||!patch)return target;
  for(const [key,value] of Object.entries(patch)){if(value!==undefined)target[key]=value;}
  if(patch.commonReportFormUrl!==undefined)target.reportFormUrl=patch.commonReportFormUrl;
  if(patch.activityType!==undefined)target.type=patch.activityType;
  if(patch.nightLinked!==undefined)target.affectsAttendance=Boolean(patch.nightLinked);
  return target;
}

async function migrateAdminProgramOverridesToSheetOnce(){
  if(currentRoleId()!=="admin" || state.settings?.programSheetMigrationV07662 || !window.schoolBoard?.saveProgramOperation) return;
  const overrides=state.programOverrides&&typeof state.programOverrides==="object"?state.programOverrides:{};
  const programs=Array.isArray(readonlyCache?.programs)?readonlyCache.programs:[];
  let changed=false, failed=false;
  for(const [id,override] of Object.entries(overrides)){
    if(!override || !["date","time","place","teacher"].some(key=>String(override[key]??"").trim())) continue;
    const program=programs.find(item=>String(item.id||"")===String(id));
    if(!program?.source) continue;
    try{
      const result=await window.schoolBoard.saveProgramOperation({role:"admin",source:program.source,programId:program.programId||program.courseId||"",sessionId:program.sessionId||"",date:override.date??program.date??"",time:override.time??program.time??"",place:override.place??program.place??"",teacher:override.teacher??program.teacher??""});
      if(!result?.ok) throw new Error(result?.reason||"시트 저장 실패");
      if(result?.programPatch)applyProgramWritePatch08172(program,result.programPatch);
      const kept={...override}; ["date","time","place","teacher"].forEach(key=>delete kept[key]);
      if(Object.keys(kept).length) state.programOverrides[id]=kept; else delete state.programOverrides[id];
      changed=true;
    }catch(error){ failed=true; console.warn("[UEP] 프로그램 로컬값 시트 이관 실패",id,error); }
  }
  if(!failed){ state.settings.programSheetMigrationV07662=true; changed=true; }
  if(changed) await save();
}

  }
  })();
