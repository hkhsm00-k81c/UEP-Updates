function normalizeHeaderText(value) {
  return String(value ?? "")
    .replace(/\r?\n/g, " ")
    .replace(/\s+/g, " ")
    .trim();
}

function detectHeaderRowIndex(rows) {
  const source = Array.isArray(rows) ? rows : [];
  const scanLimit = Math.min(source.length, 12);
  let bestIndex = -1;
  let bestScore = -Infinity;

  for (let index = 0; index < scanLimit; index += 1) {
    const cells = (source[index] || []).map(normalizeHeaderText);
    const nonEmpty = cells.filter(Boolean);
    if (nonEmpty.length < 2) continue;

    const uniqueCount = new Set(nonEmpty).size;
    const duplicatePenalty = nonEmpty.length - uniqueCount;
    const knownHeaderHits = nonEmpty.filter((value) =>
      /^(학생ID|학번|성명|사용성명|학년|반|번호|일자|운영일|일정ID|프로그램ID|강좌ID|기록ID|상태|구분|설정키|파일구분|스프레드시트ID|탭명|탭용도)$/.test(value),
    ).length;
    const titlePenalty = nonEmpty.length <= 2 && /UEP|표준|시트|DB|관리|기준/.test(nonEmpty.join(" ")) ? 8 : 0;
    const score = nonEmpty.length * 3 + uniqueCount + knownHeaderHits * 8 - duplicatePenalty * 2 - titlePenalty;

    if (score > bestScore) {
      bestScore = score;
      bestIndex = index;
    }
  }

  return bestIndex;
}

function headerLookupKey(value) {
  return normalizeHeaderText(value).replace(/[\s·_\-()\[\]{}]/g, "").toLowerCase();
}

function valueByHeader(row, ...aliases) {
  if (!row || typeof row !== "object") return "";
  for (const alias of aliases) {
    if (Object.prototype.hasOwnProperty.call(row, alias)) return row[alias];
  }
  const wanted = new Set(aliases.map(headerLookupKey));
  for (const [key, value] of Object.entries(row)) {
    const normalizedKey = headerLookupKey(key);
    if (wanted.has(normalizedKey)) return value;
    // 병합형 2단 헤더(예: "학생정보 학번", "외출정보 외출일자")도
    // 하위 헤더 별칭으로 안전하게 찾습니다.
    for (const aliasKey of wanted) {
      if (aliasKey && normalizedKey.endsWith(aliasKey)) return value;
    }
  }
  return "";
}

function looksLikeSubHeaderRow(row) {
  const cells = (Array.isArray(row) ? row : []).map(normalizeHeaderText);
  const nonEmpty = cells.filter(Boolean);
  if (nonEmpty.length < 2) return false;
  const headerTerms = nonEmpty.filter((value) =>
    /^(?:원점수|표준점수|백분위|등급|과목|학생ID|학번|성명|이름|시험명|시험월|시험년도|학년도|반|번호|일자|날짜|외출일자|외출시간|복귀예정시간|귀사예정시간|자택도착예정시간|사유|목적지|구분|상태|처리상태|승인자|비고)$/.test(value),
  ).length;
  const numericLike = nonEmpty.filter((value) => /^[-+]?\d+(?:[.,]\d+)?$/.test(value)).length;
  return headerTerms >= 2 || (headerTerms >= 1 && numericLike === 0);
}

function composeTwoLevelHeaders(parentRow, childRow) {
  const parents = (Array.isArray(parentRow) ? parentRow : []).map(normalizeHeaderText);
  const children = (Array.isArray(childRow) ? childRow : []).map(normalizeHeaderText);
  const width = Math.max(parents.length, children.length);
  const result = [];
  let activeParent = "";
  for (let index = 0; index < width; index += 1) {
    const parent = parents[index] || "";
    const child = children[index] || "";
    if (parent) activeParent = parent;
    if (child) {
      const parentKey = headerLookupKey(activeParent);
      const childKey = headerLookupKey(child);
      result.push(parentKey && parentKey !== childKey ? `${activeParent} ${child}`.trim() : child);
    } else {
      result.push(parent || activeParent || `__EMPTY_${index + 1}`);
    }
  }
  return result;
}

function uniqueHeaders(headers) {
  const seen = new Map();
  return headers.map((header, index) => {
    const base = normalizeHeaderText(header) || `__EMPTY_${index + 1}`;
    const count = (seen.get(base) || 0) + 1;
    seen.set(base, count);
    return count === 1 ? base : `${base}__${count}`;
  });
}

function applyCanonicalHeaderAliases(row) {
  if (!row || typeof row !== "object") return row;
  const result = { ...row };
  for (const [canonical, aliases] of Object.entries(CANONICAL_HEADER_ALIASES)) {
    if (String(result[canonical] ?? "").trim()) continue;
    const value = valueByHeader(result, canonical, ...aliases);
    if (value !== "" && value != null) result[canonical] = value;
  }
  return result;
}

function matrixToObjectRows(matrix) {
  const rows = Array.isArray(matrix) ? matrix : [];
  const headerIndex = detectHeaderRowIndex(rows);
  if (headerIndex < 0) return [];

  let dataStart = headerIndex + 1;
  let header = (rows[headerIndex] || []).map(normalizeHeaderText);
  const nextRow = rows[headerIndex + 1] || [];
  const parentHasGroups = header.some(Boolean) && (header.filter(Boolean).length < Math.max(header.length, nextRow.length) * 0.8 || new Set(header.filter(Boolean)).size < header.filter(Boolean).length);
  if (looksLikeSubHeaderRow(nextRow) && parentHasGroups) {
    header = composeTwoLevelHeaders(rows[headerIndex], nextRow);
    dataStart = headerIndex + 2;
  }
  header = uniqueHeaders(header);

  return rows.slice(dataStart)
    .filter((row) => Array.isArray(row) && row.some((value) => String(value ?? "").trim()))
    .map((row) => applyCanonicalHeaderAliases(Object.fromEntries(
      header.map((key, index) => [key, row[index] ?? ""]),
    )));
}

function serialToDate(value) {
  const serial = Number(value);
  if (!Number.isFinite(serial) || serial < 20000 || serial > 80000) return "";
  const utc = new Date(Math.round((serial - 25569) * 86400 * 1000));
  return `${utc.getUTCFullYear()}-${String(utc.getUTCMonth() + 1).padStart(2, "0")}-${String(utc.getUTCDate()).padStart(2, "0")}`;
}

function localDate(value) {
  if (value === "" || value == null) return "";
  if (value instanceof Date && !Number.isNaN(value.getTime()))
    return `${value.getFullYear()}-${String(value.getMonth() + 1).padStart(2, "0")}-${String(value.getDate()).padStart(2, "0")}`;
  const serialDate = serialToDate(value);
  if (serialDate) return serialDate;
  const text = String(value).trim();
  const match = text.match(/(\d{4})[.\-/년]\s*(\d{1,2})[.\-/월]\s*(\d{1,2})/);
  if (match) return `${match[1]}-${match[2].padStart(2, "0")}-${match[3].padStart(2, "0")}`;
  const parsedText = new Date(text);
  if (!Number.isNaN(parsedText.getTime()) && /GMT|UTC|^[A-Za-z]{3}/.test(text))
    return `${parsedText.getUTCFullYear()}-${String(parsedText.getUTCMonth() + 1).padStart(2, "0")}-${String(parsedText.getUTCDate()).padStart(2, "0")}`;
  return text;
}

function localTime(value) {
  if (value === "" || value == null) return "";
  if (value instanceof Date && !Number.isNaN(value.getTime()))
    return `${String(value.getHours()).padStart(2, "0")}:${String(value.getMinutes()).padStart(2, "0")}`;
  if (typeof value === "number") {
    const totalMinutes = Math.round((value % 1) * 24 * 60);
    return `${String(Math.floor(totalMinutes / 60) % 24).padStart(2, "0")}:${String(totalMinutes % 60).padStart(2, "0")}`;
  }
  const text = String(value).trim();
  if (/^0\.\d+$/.test(text)) return localTime(Number(text));
  const match = text.match(/(\d{1,2}):(\d{2})/);
  return match ? `${match[1].padStart(2, "0")}:${match[2]}` : text;
}

// __UEP_SELECTION_TERM_08082__
function normalizeSelectionTerm(value){
  const s=String(value||'').replace(/\s+/g,'').replace(/학년도/g,'').trim();
  let m=s.match(/([23])(?:학년)?[-_.]?([12])(?:학기)?/);
  if(!m)m=s.match(/([23])학년([12])학기/);
  if(m)return `${m[1]}학년 ${m[2]}학기`;
  return String(value||'').trim();
}
