function beginCleanShutdown(reason="user-close") {
  if (shutdownStarted) return;
  shutdownStarted = true;
  try { appendShutdownLog(`shutdown-start:${reason}:pid=${process.pid}:windows=${BrowserWindow.getAllWindows().length}`); } catch {}
  destroyAuxiliaryWindows(reason);
  // Give Electron a short grace period to release renderer/GPU resources. If the
  // normal quit sequence stalls, force only THIS Electron process to exit.
  setTimeout(() => {
    try { appendShutdownLog(`shutdown-force-exit:${reason}:windows=${BrowserWindow.getAllWindows().length}`); } catch {}
    try { app.exit(0); } catch {}
  }, 1500).unref?.();
}

function parseComciganPeriodStart(label="") {
  const text=String(label||"");
  let m=text.match(/(\d{2})-(\d{2})-(\d{2})\s*[~～-]/);
  if(m) return new Date(2000+Number(m[1]),Number(m[2])-1,Number(m[3]),12,0,0);
  m=text.match(/(?:\(|^)(\d{1,2})\/(\d{1,2})\s*[~～-]/);
  if(m){const now=new Date();return new Date(now.getFullYear(),Number(m[1])-1,Number(m[2]),12,0,0);}
  return null;
}
function selectComciganPeriod(periods,targetDate){
  const target=new Date(`${targetDate}T12:00:00`);
  const list=Array.isArray(periods)?periods:[];
  for(const row of list){
    const id=Array.isArray(row)?Number(row[0]):Number(row?.id||row?.period||0);
    const label=Array.isArray(row)?row[1]:row?.label||row?.name||"";
    const start=parseComciganPeriodStart(label);
    if(!start||!id) continue;
    const end=new Date(start); end.setDate(end.getDate()+6);
    if(target>=start&&target<=end) return id;
  }
  return Number(Array.isArray(list[0])?list[0][0]:list[0]?.id||1)||1;
}

function parseOwner(text) {
  const m=String(text||"").match(/^\(([^)]+)\)/);
  const token=(m?.[1]||"").trim();
  if(!token) return ["",""];
  if(/부$|부\s/.test(token)) return [token.replace(/\s*확정$/,""),""];
  const parts=token.split(/\s+/);
  if(parts.length>1 && /부$/.test(parts[0])) return [parts[0],parts.slice(1).join(" ")];
  return ["",token.replace(/\s*확정$/," ").trim()];
}

function validateServiceAccount(value) {
  return Boolean(
    value && value.type === "service_account" &&
    typeof value.client_email === "string" &&
    typeof value.private_key === "string" &&
    value.private_key.startsWith("-----BEGIN PRIVATE KEY-----") &&
    value.token_uri === "https://oauth2.googleapis.com/token"
  );
}

function recoveryKey(salt) {
  // safeStorage가 Windows/Chromium 키 문제로 복호화에 실패할 때만 사용하는 로컬 복구키입니다.
  // 동일 Windows 사용자/PC에서 재부팅 후에도 안정적으로 재생성됩니다.
  const identity = `${os.hostname()}|${os.userInfo().username}|UEP-CREDENTIAL-RECOVERY-v1`;
  return crypto.scryptSync(identity, salt, 32);
}

async function writeCredentialRecovery(value) {
  const salt = crypto.randomBytes(16), iv = crypto.randomBytes(12);
  const key = recoveryKey(salt);
  const cipher = crypto.createCipheriv("aes-256-gcm", key, iv);
  const encrypted = Buffer.concat([cipher.update(JSON.stringify(value), "utf8"), cipher.final()]);
  const payload = {
    version: 1, algorithm: "aes-256-gcm",
    salt: salt.toString("base64"), iv: iv.toString("base64"),
    tag: cipher.getAuthTag().toString("base64"), data: encrypted.toString("base64")
  };
  await fs.mkdir(path.dirname(credentialRecoveryPath()), { recursive: true });
  await fs.writeFile(credentialRecoveryPath(), JSON.stringify(payload), "utf8");
}

async function readCredentialRecovery() {
  const payload = JSON.parse(await fs.readFile(credentialRecoveryPath(), "utf8"));
  if (!payload || payload.version !== 1 || payload.algorithm !== "aes-256-gcm") throw new Error("invalid recovery credential");
  const salt=Buffer.from(payload.salt,"base64"), iv=Buffer.from(payload.iv,"base64"), tag=Buffer.from(payload.tag,"base64"), data=Buffer.from(payload.data,"base64");
  const decipher = crypto.createDecipheriv("aes-256-gcm", recoveryKey(salt), iv);
  decipher.setAuthTag(tag);
  return JSON.parse(Buffer.concat([decipher.update(data), decipher.final()]).toString("utf8"));
}

async function saveEncrypted(filePath, value) {
  await fs.mkdir(path.dirname(filePath), { recursive: true });
  if (safeStorage.isEncryptionAvailable()) {
    await fs.writeFile(filePath, safeStorage.encryptString(JSON.stringify(value)));
  } else {
    // 인증정보는 아래 복구 저장소가 있으므로, 다른 보안값은 기존처럼 실패시킵니다.
    if (path.resolve(filePath) !== path.resolve(credentialPath())) throw new Error("Windows 보안 저장소를 사용할 수 없습니다.");
  }
  if (path.resolve(filePath) === path.resolve(credentialPath())) await writeCredentialRecovery(value);
}

async function readEncrypted(filePath) {
  let encrypted = null;
  try { encrypted = await fs.readFile(filePath); } catch (readError) {
    if (path.resolve(filePath) === path.resolve(credentialPath())) {
      try { return await readCredentialRecovery(); } catch {}
    }
    if (path.resolve(filePath) === path.resolve(credentialPath()) && (readError?.code === "ENOENT" || /no such file/i.test(String(readError?.message||"")))) {
      const friendly = new Error("담임 배포 PC의 Google 시트 연결 승인이 아직 준비되지 않았습니다.");
      friendly.code = "UEP_GOOGLE_CREDENTIAL_MISSING";
      friendly.setupRequired = true;
      throw friendly;
    }
    throw readError;
  }
  if (safeStorage.isEncryptionAvailable()) {
    try { return JSON.parse(safeStorage.decryptString(encrypted)); } catch {}
  }
  // 일부 구버전/수동 이관 파일이 평문 JSON인 경우 현재 형식으로 재저장합니다.
  try {
    const legacy = JSON.parse(encrypted.toString("utf8"));
    if (legacy && typeof legacy === "object") { await saveEncrypted(filePath, legacy); return legacy; }
  } catch {}
  // Google 서비스 계정은 safeStorage가 깨져도 동일 PC/사용자용 AES-GCM 복구본으로 자동 복원합니다.
  if (path.resolve(filePath) === path.resolve(credentialPath())) {
    try {
      const recovered = await readCredentialRecovery();
      if (validateServiceAccount(recovered)) {
        try { if (safeStorage.isEncryptionAvailable()) await fs.writeFile(filePath, safeStorage.encryptString(JSON.stringify(recovered))); } catch {}
        return recovered;
      }
    } catch {}
  }
  const backup = `${filePath}.unreadable-${Date.now()}.bak`;
  try { await fs.rename(filePath, backup); } catch {}
  const error = new Error("저장된 보안 인증정보를 복호화할 수 없습니다. 서비스 계정 JSON을 다시 등록해 주세요.");
  error.code = "UEP_SAFE_STORAGE_DECRYPT_FAILED";
  throw error;
}

function googleUserOAuthRecoveryKey(salt){
  const identity=`${os.hostname()}|${os.userInfo().username}|UEP-GOOGLE-USER-OAUTH-v1`;
  return crypto.scryptSync(identity,salt,32);
}
function encodeGoogleUserOAuthRecovery(value){
  const salt=crypto.randomBytes(16),iv=crypto.randomBytes(12),key=googleUserOAuthRecoveryKey(salt);
  const cipher=crypto.createCipheriv('aes-256-gcm',key,iv);
  const encrypted=Buffer.concat([cipher.update(JSON.stringify(value),'utf8'),cipher.final()]);
  return {version:1,algorithm:'aes-256-gcm',salt:salt.toString('base64'),iv:iv.toString('base64'),tag:cipher.getAuthTag().toString('base64'),data:encrypted.toString('base64')};
}
function decodeGoogleUserOAuthRecovery(payload){
  if(!payload||payload.version!==1||payload.algorithm!=='aes-256-gcm')throw new Error('invalid OAuth recovery token');
  const salt=Buffer.from(payload.salt,'base64'),iv=Buffer.from(payload.iv,'base64'),tag=Buffer.from(payload.tag,'base64'),data=Buffer.from(payload.data,'base64');
  const decipher=crypto.createDecipheriv('aes-256-gcm',googleUserOAuthRecoveryKey(salt),iv);decipher.setAuthTag(tag);
  return JSON.parse(Buffer.concat([decipher.update(data),decipher.final()]).toString('utf8'));
}

async function getSheetsToken(credentials) {
  const now = Math.floor(Date.now() / 1000);
  const header = base64url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claims = base64url(JSON.stringify({
    iss: credentials.client_email,
    scope: "https://www.googleapis.com/auth/spreadsheets",
    aud: credentials.token_uri,
    iat: now,
    exp: now + 3600,
  }));
  const unsigned = `${header}.${claims}`;
  const signature = crypto.sign("RSA-SHA256", Buffer.from(unsigned), credentials.private_key).toString("base64url");
  const body = new URLSearchParams({
    grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
    assertion: `${unsigned}.${signature}`,
  });
  const response = await fetch(credentials.token_uri, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body,
  });
  if (!response.ok) throw new Error(`Google 인증 실패 (${response.status})`);
  const payload = await response.json();
  if (!payload.access_token) throw new Error("Google 인증 토큰을 받지 못했습니다.");
  return payload.access_token;
}

async function readSheetBatch(token, spreadsheetId, ranges, attempt = 0) {

  url.searchParams.set("dateTimeRenderOption", "SERIAL_NUMBER");
  ranges.forEach((range) => url.searchParams.append("ranges", range));
  const response = await fetch(url, { headers: { authorization: `Bearer ${token}` } });
  if (!response.ok) {
    let detail = "";
    try { detail = (await response.json())?.error?.message || ""; } catch {}
    if ((response.status === 429 || response.status >= 500) && attempt < 3) {
      const retryAfter = Number(response.headers.get("retry-after") || 0);
      const waitMs = retryAfter > 0 ? retryAfter * 1000 : 900 * Math.pow(2, attempt);
      await new Promise(resolve => setTimeout(resolve, waitMs));
      return readSheetBatch(token, spreadsheetId, ranges, attempt + 1);
    }
    throw new Error(`UEP 구글시트 읽기 실패 (${response.status})${detail ? `: ${detail}` : ""}`);
  }
  return (await response.json()).valueRanges || [];
}

function columnLetter(index) {
  let n=Number(index)+1, out="";
  while(n>0){ const r=(n-1)%26; out=String.fromCharCode(65+r)+out; n=Math.floor((n-1)/26); }
  return out;
}
function normalizeSheetHeader(value){return String(value??"").replace(/\r?\n/g," ").replace(/\s+/g," ").trim();}
function findHeaderRowByRequired(matrix,required){
  const wanted=required.map(x=>String(x));
  const limit=Math.min(Array.isArray(matrix)?matrix.length:0,12);
  for(let i=0;i<limit;i+=1){
    const row=(matrix[i]||[]).map(normalizeSheetHeader);
    if(wanted.every(name=>row.includes(name))) return {index:i,headers:row};
  }
  return null;
}
async function saveStudentContact(payload={}) {
  const allowed={phone:["학생연락처"],email:["학생이메일","이메일","이메일주소"],guardian1Phone:["보호자1연락처"],guardian2Phone:["보호자2연락처"]};
  const field=String(payload.field||"");
  if(!allowed[field]) throw new Error("수정할 수 없는 학생정보 항목입니다.");
  const role=String(payload.role||"");
  if(!["admin","homeroom"].includes(role)) throw new Error("학생 연락정보 수정 권한이 없습니다.");
  const account=await readEncrypted(credentialPath());
  if(!validateServiceAccount(account)) throw new Error("저장된 Google 서비스 계정 인증정보가 올바르지 않습니다.");
  const token=await getSheetsToken(account);
  const range="'04_학생연락식별정보'!A1:Z1000";
  const result=await readSheetBatch(token,UEP_SPREADSHEET_ID,[range]);
  const matrix=result?.[0]?.values||[];
  const header=findHeaderRowByRequired(matrix,["학생ID"]);
  if(!header) throw new Error("04_학생연락식별정보의 헤더에서 학생ID를 찾지 못했습니다.");
  const studentId=String(payload.studentId||"").trim();
  if(!studentId) throw new Error("학생ID가 없어 저장할 수 없습니다.");
  const idCol=header.headers.indexOf("학생ID");
  let fieldCol=-1;
  for(const name of allowed[field]){ const idx=header.headers.indexOf(name); if(idx>=0){fieldCol=idx;break;} }
  if(fieldCol<0) throw new Error(`04_학생연락식별정보에 ${allowed[field][0]} 열이 없습니다.`);
  let dataRow=-1;
  for(let r=header.index+1;r<matrix.length;r+=1){ if(String(matrix[r]?.[idCol]||"").trim()===studentId){dataRow=r;break;} }
  const value=String(payload.value??"").trim();
  if(dataRow<0){
    const row=Array(Math.max(header.headers.length,fieldCol+1)).fill("");
    row[idCol]=studentId; row[fieldCol]=value;
    const useCol=header.headers.indexOf("현재사용여부"); if(useCol>=0)row[useCol]="Y";
    await appendSheetValues(token,UEP_SPREADSHEET_ID,"'04_학생연락식별정보'!A:Z",[row]);
  }else{
    const sheetRow=dataRow+1;
    const col=columnLetter(fieldCol);
    await updateSheetValues(token,UEP_SPREADSHEET_ID,`'04_학생연락식별정보'!${col}${sheetRow}:${col}${sheetRow}`,[[value]]);
  }
  liveDataCache=null; liveDataFetchedAt=0;
  const data=await fetchLiveData({force:true});
  return {ok:true,data};
}

async function clearSheetValues(token, spreadsheetId, range) {
  const url = `https://sheets.googleapis.com/v4/spreadsheets/${spreadsheetId}/values/${encodeURIComponent(range)}:clear`;
  const response = await fetch(url, {
    method: "POST",
    headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },
    body: "{}",
  });
  if (!response.ok) throw new Error(`공결 삭제 실패 (${response.status})`);
  return response.json();
}

function dormRuleDateValue(value){
  if(value===null||value===undefined||value==='')return '';
  const raw=String(value).trim();
  if(/^\d+(?:\.\d+)?$/.test(raw)){
    const serial=Number(raw);
    if(Number.isFinite(serial)&&serial>20000&&serial<100000){
      const d=new Date(Date.UTC(1899,11,30)+Math.round(serial*86400000));
      if(!Number.isNaN(d.getTime())){
        const y=d.getUTCFullYear(),mo=String(d.getUTCMonth()+1).padStart(2,'0'),day=String(d.getUTCDate()).padStart(2,'0');
        const hh=String(d.getUTCHours()).padStart(2,'0'),mm=String(d.getUTCMinutes()).padStart(2,'0');
        return `${y}-${mo}-${day}T${hh}:${mm}`;
      }
    }
  }
  const mm=raw.replace(' ','T').match(/^(\d{4})-(\d{1,2})-(\d{1,2})T(\d{1,2}):(\d{2})/);
  return mm?`${mm[1]}-${String(mm[2]).padStart(2,'0')}-${String(mm[3]).padStart(2,'0')}T${String(mm[4]).padStart(2,'0')}:${mm[5]}`:raw;
}

async function appendSheetValues(token, spreadsheetId, range, values) {
  const url = new URL(`https://sheets.googleapis.com/v4/spreadsheets/${spreadsheetId}/values/${encodeURIComponent(range)}:append`);
  url.searchParams.set("valueInputOption", "USER_ENTERED");
  url.searchParams.set("insertDataOption", "INSERT_ROWS");
  const response = await fetch(url, {
    method: "POST",
    headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json" },

  });

async function mapWithConcurrency(items, limit, worker) {
  const queue = Array.from(items || []);
  const results = new Array(queue.length);
  let cursor = 0;
  const runners = Array.from({ length: Math.max(1, Math.min(limit, queue.length || 1)) }, async () => {
    while (true) {
      const index = cursor++;
      if (index >= queue.length) return;
      results[index] = await worker(queue[index], index);
    }
  });
  await Promise.all(runners);
  return results;
}

function ymd(value = new Date()) {
  const date = value instanceof Date ? value : new Date(value);
  return `${date.getFullYear()}${String(date.getMonth() + 1).padStart(2, "0")}${String(date.getDate()).padStart(2, "0")}`;
}

function mondayOf(dateValue) {
  const date = dateValue instanceof Date ? new Date(dateValue) : new Date(dateValue || Date.now());
  date.setHours(12, 0, 0, 0);
  const day = date.getDay();
  date.setDate(date.getDate() - (day === 0 ? 6 : day - 1));
  return date;
}

async function fetchJson(url, timeoutMs = 12000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  try {
    const response = await fetch(url, {
      signal: controller.signal,
      headers: { "user-agent": "UEP/0.42.7 School Operations" },
    });
    if (!response.ok) throw new Error(`공공데이터 연결 실패 (${response.status})`);
    return await response.json();
  } finally {
    clearTimeout(timer);
  }
}

function neisRows(payload, key) {
  const group = payload?.[key];
  if (!Array.isArray(group)) return [];
  const rowBlock = group.find((item) => Array.isArray(item?.row));
  return rowBlock?.row || [];
}

function neisResultMessage(payload, key) {
  const result = payload?.RESULT || payload?.[key]?.[0]?.head?.find?.((item) => item.RESULT)?.RESULT;
  if (!result) return "";
  return result.MESSAGE || result.CODE || "";
}

function neisUrl(dataset, params, apiKey = "") {
  const url = new URL(`https://open.neis.go.kr/hub/${dataset}`);
  url.searchParams.set("Type", "json");
  url.searchParams.set("pIndex", "1");
  url.searchParams.set("pSize", "100");
  if (apiKey) url.searchParams.set("KEY", apiKey);
  Object.entries(params || {}).forEach(([key, value]) => {
    if (value !== "" && value !== null && value !== undefined) url.searchParams.set(key, String(value));
  });
  return url.toString();
}

async function findNeisSchool({ schoolName, officeCode = "", schoolCode = "", apiKey = "" }) {
  if (officeCode && schoolCode) return { schoolName, officeCode, schoolCode };
  const payload = await fetchJson(neisUrl("schoolInfo", { SCHUL_NM: schoolName }, apiKey));
  const rows = neisRows(payload, "schoolInfo");
  const exact = rows.find((row) => String(row.SCHUL_NM || "").replace(/\s/g, "") === String(schoolName || "").replace(/\s/g, ""));
  const row = exact || rows[0];
  if (!row) throw new Error(neisResultMessage(payload, "schoolInfo") || `${schoolName} 학교정보를 찾지 못했습니다.`);
  return {
    schoolName: row.SCHUL_NM || schoolName,
    officeCode: row.ATPT_OFCDC_SC_CODE,
    schoolCode: row.SD_SCHUL_CODE,
    officeName: row.ATPT_OFCDC_SC_NM || "",
    address: row.ORG_RDNMA || row.ORG_RDNZC || "",
  };
}

function cleanMealText(value) {
  return String(value || "")
    .replace(/<br\s*\/?\s*>/gi, "\n")
    .replace(/\([^)]*\)/g, (match) => match)
    .split(/\n+/)
    .map((item) => item.trim())
    .filter(Boolean);
}

function neisDateToLocalDate(value) {
  const text = String(value || "").replace(/\D/g, "");
  if (!/^\d{8}$/.test(text)) return null;
  const date = new Date(Number(text.slice(0,4)), Number(text.slice(4,6))-1, Number(text.slice(6,8)), 12, 0, 0);
  return Number.isNaN(date.getTime()) ? null : date;
}

function expectedSchoolPeriods(value) {
  const date = value instanceof Date ? value : neisDateToLocalDate(value);
  const weekdayNo = date?.getDay();
  if (weekdayNo === 3) return 5;
  return [1,2,4,5].includes(weekdayNo) ? 7 : 0;
}

    const mergedByPeriod = new Map();
    [...fallbackLessons, ...exactLessons]
      .filter((item) => item.period > 0)
      .sort((a, b) => a.period - b.period)
      .forEach((item) => {
        const current = mergedByPeriod.get(item.period);
        if (!current || current.subject === "-" || item.source === "학급 직접조회") mergedByPeriod.set(item.period, item);
      });
    const rawLessons = [...mergedByPeriod.values()].sort((a, b) => a.period - b.period);
    const directPeriods = new Set(exactLessons.map((item) => item.period).filter(Boolean));
    const recoveredPeriods = rawLessons.filter((item) => item.source === "1~9반 교차조회" && !directPeriods.has(item.period)).map((item) => item.period);
    return {
      date: ymd(date),
      weekday: ["일", "월", "화", "수", "목", "금", "토"][date.getDay()],
      error: exactVariants.map((entry) => entry?.payload?.__error).filter(Boolean).join(" / "),
      rawRowCount: exactRows.length,
      gradeWideRowCount: dedupedRows.length,
      gradeMatchedRowCount: matchingRows.length,
      gradeClassValues: classValues,
      recoveredPeriods,
      rawPeriods: rawLessons.map((item) => item.period),
      rawLessons,
      rawResponse: {
        exact: exactVariants.map(x=>({mode:x.mode,url:String(x.url||"").replace(/([?&]KEY=)[^&]+/i,"$1***"),payload:x.payload})),
        classSweep: sweepVariants.map(x=>({mode:x.mode,classNo:x.candidate,url:String(x.url||"").replace(/([?&]KEY=)[^&]+/i,"$1***"),payload:x.payload}))
      },
      exactUrl: String(pair.exactUrl || "").replace(/([?&]KEY=)[^&]+/i,"$1***"),
      gradeUrl: "1~9반 개별 정식 조회",
      lessons: rawLessons,
    };
  });
  const diagnostics = days.map((day) => {
    const expected = expectedSchoolPeriods(day.date);
    const periods = day.lessons.map((lesson) => Number(lesson.period || 0)).filter(Boolean);
    const maxPeriod = periods.length ? Math.max(...periods) : 0;
    return {
      date: day.date,
      weekday: day.weekday,
      rowCount: day.lessons.length,
      maxPeriod,
      expected,
      complete: !day.lessons.length || maxPeriod >= expected,
      missingPeriods: expected ? Array.from({length: expected}, (_,i)=>i+1).filter(period=>!periods.includes(period)) : [],
      recoveredPeriods: day.recoveredPeriods || [],
      exactRowCount: day.rawRowCount || 0,
      gradeWideRowCount: day.gradeWideRowCount || 0,
      gradeMatchedRowCount: day.gradeMatchedRowCount || 0,
      error: day.error || "",
    };
  });
  return { year, semester, monday: ymd(monday), days, diagnostics, queryMode: "ALL_TI_YMD + TI_FROM_YMD/TI_TO_YMD 이중조회 · 1~9반 교차검증" };
}

async function readState() {

  catch { return null; }

let stateWriteChain = Promise.resolve();
async function writeStateNow(value) {
  const target = statePath();
  const dir = path.dirname(target);
  await fs.mkdir(dir, { recursive: true });

  // Each save gets its own temporary file. A fixed ".tmp" name can be
  // removed by a concurrent save, producing ENOENT during copy/rename.
  const temp = path.join(dir, `school-board-state.${process.pid}.${Date.now()}.${crypto.randomBytes(4).toString("hex")}.tmp`);
  const payload = JSON.stringify(value, null, 2);
  try {
    await fs.writeFile(temp, payload, "utf8");
    try {
      await fs.rename(temp, target);
    } catch (renameError) {
      // Windows may reject replacement by rename when the destination exists.
      // Fall back to a direct write; this keeps user data available instead
      // of failing the entire state:save IPC call.
      await fs.writeFile(target, payload, "utf8");
      try { await fs.unlink(temp); } catch {}
    }
  } finally {
    try { await fs.unlink(temp); } catch {}
  }
  return { ok: true, savedAt: new Date().toISOString() };
}
function writeState(value) {
  // Serialize renderer saves so rapid UI updates cannot race each other.
  const task = stateWriteChain.then(() => writeStateNow(value));
  stateWriteChain = task.catch(() => {});
  return task;
}

    // 기존 입력센터 제출 갱신도 한 번의 batchUpdate로 처리하여 Write quota를 소모하지 않습니다.
    const reportUpdates=actions.filter(a=>a.type==='update').map(a=>({range:`'${targetSheet}'!A${a.sheetRow}:X${a.sheetRow}`,values:[a.row]}));
    await batchUpdateSheetValues(token,UEP_PROCESSING_SPREADSHEET_ID,reportUpdates);
    // 신규 입력센터 정규화는 출처별 물리 구역을 나누지 않고 실제 마지막 데이터 행 바로 다음에 연속 저장합니다.
    const lastUsed=existing.length?Math.max(...existing.map(item=>item.sheetRow)):DATA_START_ROW-1;
    const news=actions.filter(a=>a.type==='new');
    if(news.length){
      const start=Math.max(DATA_START_ROW,lastUsed+1),end=start+news.length-1;
      if(end>INPUT_END_ROW)throw new Error(`보고서 정규화 시트의 저장 가능 행을 초과했습니다. (${DATA_START_ROW}~${INPUT_END_ROW}행)`);
      await updateSheetValues(token,UEP_PROCESSING_SPREADSHEET_ID,`'${targetSheet}'!A${start}:X${end}`,news.map(a=>a.row));
    }
    // 입력센터 원본에서 삭제된 자료는 정규화 원장에서 상태만 '삭제됨'으로 남겨 이력을 보존합니다.
    const srcResult=await readSheetBatch(token,UEP_INPUT_CENTER_SPREADSHEET_ID,["'02_자료원본'!A3:X10000"]);
    const src=(srcResult?.[0]?.values||[]).slice(1);const liveSourceIds=new Set(src.map(r=>String(r?.[0]||'')).filter(Boolean));
    const deleteWrites=[];
    for(const item of existing){
      const row=item.row;if(String(row?.[17]||'')!=='INPUT_CENTER')continue;
      const sid=String(row?.[18]||'');if(!sid||liveSourceIds.has(sid)||String(row?.[22]||'')==='삭제됨')continue;
      deleteWrites.push({range:`'${targetSheet}'!W${item.sheetRow}:W${item.sheetRow}`,values:[['삭제됨']]});
    }
    await batchUpdateSheetValues(token,UEP_PROCESSING_SPREADSHEET_ID,deleteWrites);

    // v0.80.45: 쓰기 응답만 믿지 않고 목적지 원장을 다시 읽어 실제 저장을 검증합니다.
    // 같은 원본ID가 있거나, 동일 논리키+내용해시가 실제로 존재해야만 해당 원본을 완료 처리합니다.
    const verifyResult=await readSheetBatch(token,UEP_PROCESSING_SPREADSHEET_ID,[`'${targetSheet}'!A3:X${INPUT_END_ROW}`]);
    const verifyRows=(verifyResult?.[0]?.values||[]).slice(1).filter(row=>(row||[]).some(v=>String(v??'').trim()));
    const actionVerified=a=>verifyRows.some(r=>{
      if(String(r?.[22]||'정상')==='삭제됨')return false;
      const sameSource=String(r?.[17]||'')===String(a.row?.[17]||'')&&String(r?.[18]||'')===String(a.row?.[18]||'');
      const sameLogicalContent=String(r?.[20]||'')===String(a.row?.[20]||'')&&canonicalExistingHash(r)===String(a.row?.[23]||'');
      return sameSource||sameLogicalContent;
    });
    const unverified=actions.filter(a=>!actionVerified(a));
    if(unverified.length){
      const sample=unverified.slice(0,5).map(a=>String(a.row?.[18]||a.rec?.id||'원본ID없음')).join(', ');
      throw new Error(`정규화 목적지 저장 검증 실패 ${unverified.length}건${sample?` (${sample}${unverified.length>5?', …':''})`:''}. 입력센터 원본은 완료 처리하지 않았습니다.`);
    }

    // 실제 목적지에 존재하는 준비건만 원본 처리상태를 완료합니다.
    // 본문이 없어 prepared에서 제외된 원본까지 완료되는 기존 문제도 함께 차단합니다.
    const verifiedIds=new Set(actions.map(a=>String(a.rec?.id||'')).filter(Boolean));const statusWrites=[];
    for(let i=0;i<src.length;i++)if(verifiedIds.has(String(src[i]?.[0]||'')))statusWrites.push({range:`'02_자료원본'!R${i+4}:R${i+4}`,values:[['정규화완료']]});
    await batchUpdateSheetValues(token,UEP_INPUT_CENTER_SPREADSHEET_ID,statusWrites);
    liveDataCache=null;liveDataFetchedAt=0;return {ok:true,targetSheet,programName,newCount,duplicateCount,updateCount,errorCount:0,count:newCount+updateCount,verifiedCount:actions.length};
  }catch(error){return {ok:false,reason:error?.message||String(error)};}
}

  ipcMain.handle("app:checkUpdate", async (_event, payload = {}) => {
    try {
      const rawUrl=String(payload.url||"").trim();
      if (!/^https:\/\//i.test(rawUrl)) return { ok:false, reason:"HTTPS 업데이트 정보 주소가 필요합니다." };

      // Always resolve the freshest manifest we can find. During school-wide use,
      // raw.githubusercontent.com / CDN caches can be stale or rate-limited, so:
      //   1) derive GitHub Contents API + raw + jsDelivr candidates,
      //   2) cache-bust non-API candidates,
      //   3) read every reachable candidate and choose the HIGHEST semantic version.
      const stamp=Date.now();
      const candidates=[];
      const push=(url,kind)=>{if(url && !candidates.some(x=>x.url===url))candidates.push({url,kind});};
      let canonical=rawUrl;
      try {
        const u=new URL(rawUrl);
        u.search=""; u.hash="";
        canonical=u.toString();
        if(u.hostname.toLowerCase()==="raw.githubusercontent.com"){
          const parts=u.pathname.split('/').filter(Boolean);
          if(parts.length>=4){
            const [owner,repo,ref,...fileParts]=parts;
            const filePath=fileParts.join('/');
            push(`https://api.github.com/repos/${owner}/${repo}/contents/${filePath}?ref=${encodeURIComponent(ref)}`,'github-api');
            push(`${u.toString()}?uep=${stamp}`,'github-raw');
            push(`https://cdn.jsdelivr.net/gh/${owner}/${repo}@${ref}/${filePath}?uep=${stamp}`,'jsdelivr');
          }
        } else {
          push(`${u.toString()}${u.search?'&':'?'}uep=${stamp}`,'direct');
        }
      } catch {
        push(`${rawUrl}${rawUrl.includes('?')?'&':'?'}uep=${stamp}`,'direct');
      }
      if(!candidates.length) push(canonical,'direct');

      const parseVersion=v=>String(v||'').trim().split('.').slice(0,3).map(x=>Number(String(x).replace(/\D.*$/,''))||0);
      const cmpVersion=(va,vb)=>{const a=parseVersion(va),b=parseVersion(vb);for(let i=0;i<3;i++){if((a[i]||0)!==(b[i]||0))return (a[i]||0)-(b[i]||0);}return 0;};
      const successes=[];
      let lastStatus=0,lastError='';

      for(const candidate of candidates){
        try{
          const isApi=candidate.kind==='github-api';
          const response=await net.fetch(candidate.url,{headers:{
            "user-agent":`UEP/${app.getVersion()} UpdateCheck`,
            "cache-control":"no-cache, no-store, max-age=0",
            "pragma":"no-cache",
            ...(isApi?{"accept":"application/vnd.github+json"}:{})
          }});
          lastStatus=response.status;
          if(!response.ok){lastError=`${candidate.kind}: HTTP ${response.status}`;continue;}
          let data;
          if(isApi){
            const wrapper=await response.json();
            if(!wrapper?.content){lastError='github-api: content 없음';continue;}
            data=JSON.parse(Buffer.from(String(wrapper.content).replace(/\s/g,''),'base64').toString('utf8'));
          }else data=await response.json();
          const version=String(data?.version||data?.latestVersion||'').trim();
          if(!/^\d+\.\d+\.\d+/.test(version)){lastError=`${candidate.kind}: version 없음`;continue;}
          successes.push({data,version,source:candidate.kind,url:candidate.url});
        }catch(error){lastError=`${candidate.kind}: ${String(error?.message||error)}`;}
      }

      if(!successes.length) return {ok:false,reason:`업데이트 정보 읽기 실패${lastStatus?` (${lastStatus})`:''}${lastError?` · ${lastError}`:''}`};
      successes.sort((a,b)=>cmpVersion(b.version,a.version));
      const chosen=successes[0];
      const data=chosen.data;
      const version=chosen.version;
      const current=String(payload.currentVersion||"0.0.0");
      const newer=cmpVersion(version,current)>0;
      return {
        ok:true,
        version,
        updateAvailable:newer,
        downloadUrl:String(data.downloadUrl||data.url||""),
        sha256:String(data.sha256||data.sha256sum||"").trim().toLowerCase(),
        notes:String(data.notes||data.releaseNotes||""),
        source:chosen.source,
        checkedVersions:successes.map(x=>`${x.source}:${x.version}`)
      };
    } catch(error){return {ok:false,reason:error?.message||"업데이트 확인 중 오류가 발생했습니다."};}
  });

// Recover only credentials already provisioned on this Windows user/PC.
// Nothing is embedded in the app and no personal Google OAuth token is required.
function schoolConnectionLegacyRoots(){
  const appData=process.env.APPDATA||app.getPath("appData");
  return [...new Set([
    stableUserDataRoot(),legacyUserDataRoot(),
    path.join(appData,"UNHO Education Platform"),
    path.join(appData,"unho-education-platform"),
    path.join(appData,"UEP"),
    path.join(appData,"UEP School Board"),
    path.join(appData,"School Board")
  ].filter(Boolean).map(value=>path.resolve(value)))];
}
async function readCredentialRecoveryFrom(filePath){
  const payload=JSON.parse(await fs.readFile(filePath,"utf8"));
  if(!payload||payload.version!==1||payload.algorithm!=="aes-256-gcm")throw new Error("invalid recovery credential");
  const salt=Buffer.from(payload.salt,"base64"),iv=Buffer.from(payload.iv,"base64"),tag=Buffer.from(payload.tag,"base64"),data=Buffer.from(payload.data,"base64");
  const decipher=crypto.createDecipheriv("aes-256-gcm",recoveryKey(salt),iv);decipher.setAuthTag(tag);
  return JSON.parse(Buffer.concat([decipher.update(data),decipher.final()]).toString("utf8"));
}
async function resolveSchoolServiceAccount(){
  try{const current=await readEncrypted(credentialPath());if(validateServiceAccount(current))return current;}catch{}
  for(const root of schoolConnectionLegacyRoots()){
    for(const fileName of ["google-service-account.recovery.json","google-service-account.json"]){
      const candidate=path.join(root,fileName);
      try{
        const recovered=fileName.endsWith("recovery.json")?await readCredentialRecoveryFrom(candidate):JSON.parse(await fs.readFile(candidate,"utf8"));
        if(!validateServiceAccount(recovered))continue;
        await saveEncrypted(credentialPath(),recovered);
        console.log(`[UEP] 학교 공용 Google 연결을 기존 UEP 저장소에서 복구했습니다: ${path.basename(root)}`);
        return recovered;
      }catch{}
    }
  }
  const error=new Error("이 PC에서 기존 학교 공용 Google 연결을 찾지 못했습니다. 관리자 배포본의 공용 연결 복구 상태를 확인해 주세요.");
  error.code="UEP_SCHOOL_CONNECTION_NOT_PROVISIONED";throw error;
}
