function checkPeriodAlerts() {
  const setting = state.settings.widgetSettings.timetable || {};
  if (!setting.startAlert && !setting.endAlert) return;
  const now = new Date();
  const hhmm = `${String(now.getHours()).padStart(2, "0")}:${String(now.getMinutes()).padStart(2, "0")}`;
  const key = `${dateKey(now)}-${hhmm}`;
  if (key === lastPeriodAlertKey) return;
  for (const row of (state.settings.schoolHours || [])) {
    const name = row[0], start = row[1], end = row[2];
    let message = "";
    if (setting.startAlert && start === hhmm && String(name).includes("교시")) message = `${name} 수업이 시작됩니다.`;
    if (setting.endAlert && end === hhmm && String(name).includes("교시")) message = `${name} 수업이 종료됩니다.`;
    if (message) {
      lastPeriodAlertKey = key;
      toast(message);
      try { if ("Notification" in window && Notification.permission === "granted") new Notification("UEP 시간표 알림", { body: message }); } catch {}
      break;
    }
  }
}

    clauses.push(`${label}에 대한 탐구 내용을 자신의 관심 분야와 연결하여 진로 탐색의 방향을 구체화함`);
  }
  if(follow){
    if(/도움|질병|치료|해결|피해|개선/.test(follow))clauses.push(`활동에서 생긴 질문을 바탕으로 기술이 실제 문제 해결에 활용되는 조건과 한계를 더 알아보고자 탐구 범위를 확장함`);
    else if(/찾|검색|조사|알아/.test(follow))clauses.push(`활동 후 궁금한 내용을 추가로 조사하며 탐구를 이어감`);
  }
  if(growth&&/이해|알게|처음|어려|검색|찾아|해결|성장|변화/.test(growth))clauses.push(`이해가 부족했던 내용을 이전 학습과 추가 자료 탐색을 통해 보완하며 자신의 이해 과정을 구체화함`);
  if(area==="behavior"&&/봉사|임원|회장|반장|실장|도우미/.test(`${row.title||""} ${full}`))clauses.push("공동체 활동에서 맡은 역할을 수행하며 구성원과 함께 활동하는 모습을 보임");
  return [...new Set(clauses)].slice(0,3);
}
function studentCenteredRecordPiece(row,area){
  const heading=recordProgramHeading(row);
  if(row.report){
    const clauses=reportInterpretationClauses(row,area);
    if(clauses.length)return enforceTeacherEnding(`${heading}에 참여하여 ${clauses.join(". ")}`);
  }
  if(row.kind==="activity")return enforceTeacherEnding(`${heading}에서 공동체의 일원으로 확인되는 활동과 역할을 수행함`);
  return enforceTeacherEnding(`${heading}에 참여하여 확인 가능한 활동 내용을 바탕으로 교육활동에 참여함`);
}

function buildMultiRecordDraft(rows,area,limit){
  if(!rows.length)return "";
  const pieces=rows.map(row=>studentCenteredRecordPiece(row,area));
  let result=pieces.join(" ");
  result=enforceTeacherEnding(result);
  if(result.length<=limit)return result;
  const per=Math.max(60,Math.floor((limit-(rows.length-1))/rows.length));
  result=pieces.map(piece=>piece.length>per?enforceTeacherEnding(piece.slice(0,Math.max(20,per-16)).replace(/\s+\S*$/, "")):piece).join(" ");
  if(result.length>limit)result=result.slice(0,Math.max(20,limit-16)).replace(/\s+\S*$/, "");
  return enforceTeacherEnding(result).slice(0,limit);
}
