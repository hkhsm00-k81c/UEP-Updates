function inputCenterNatureToStoredType(nature=""){
  const n=String(nature||"");
  if(n==="보고서")return "보고서·탐구보고서";
  if(n==="신청")return "프로그램 신청서";
  if(n==="만족도조사")return "만족도·설문조사";
  if(n==="출결·명단")return "출결·참여명단";
  return n||"기타";
}

function inputCenterSemester(value="",submittedAt=""){

}
