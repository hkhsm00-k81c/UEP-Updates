function selectStudentDashboardById(studentId){
  studentSensitiveRevealKeys.clear();
  const students=readonlyCache?.students||[];
  const key=String(studentId||'').trim();
  const idx=students.findIndex(student=>String(student.id||'')===key || String(student.studentNo||'')===key);
  if(idx<0)return false;
  studentDashboardIndex=idx;
  selectedStudentId=students[idx].id||students[idx].studentNo||"";
  studentPageMode='dashboard';
  window.__uepStudentDashboardSwitchToken=(window.__uepStudentDashboardSwitchToken||0)+1;
  return true;
}
