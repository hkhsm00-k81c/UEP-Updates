function dormRuleDateValue(value){
  if(value===null||value===undefined||value==='')return '';
  const raw=String(value).trim();
  if(/^\d+(?:\.\d+)?$/.test(raw)){
    const serial=Number(raw);
    if(Number.isFinite(serial)&&serial>20000&&serial<100000){
      const d=new Date(Date.UTC(1899,11,30)+Math.round(serial*86400000));
      if(!Number.isNaN(d.getTime())){
        const y=d.getUTCFullYear(),mo=String(d.getUTCMonth()+1).padStart(2,'0'),day=String(d.getUTCDate()).padStart(2,'0');

      }
    }

  const mm=raw.replace(' ','T').match(/^(\d{4})-(\d{1,2})-(\d{1,2})T(\d{1,2}):(\d{2})/);

}

async function getSheetsToken(credentials) {
  const now = Math.floor(Date.now() / 1000);
  const header = base64url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claims = base64url(JSON.stringify({
    iss: credentials.client_email,
    scope: "https://www.googleapis.com/auth/spreadsheets",
    aud: credentials.token_uri,
    iat: now,
    exp: now + 3600,
  }));
  const unsigned = `${header}.${claims}`;
  const signature = crypto.sign("RSA-SHA256", Buffer.from(unsigned), credentials.private_key).toString("base64url");
  const body = new URLSearchParams({
    grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
    assertion: `${unsigned}.${signature}`,
  });
  const response = await fetch(credentials.token_uri, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body,
  });
  if (!response.ok) throw new Error(`Google 인증 실패 (${response.status})`);
  const payload = await response.json();
  if (!payload.access_token) throw new Error("Google 인증 토큰을 받지 못했습니다.");
  return payload.access_token;
}

async function readSheetBatch(token, spreadsheetId, ranges, attempt = 0) {
  if(String(token||'').startsWith('UEP_SCHOOL_READ:')) return schoolReadBatchRead(spreadsheetId,ranges);
  const url = new URL(`https://sheets.googleapis.com/v4/spreadsheets/${spreadsheetId}/values:batchGet`);
  url.searchParams.set("majorDimension", "ROWS");
  url.searchParams.set("valueRenderOption", "UNFORMATTED_VALUE");
  url.searchParams.set("dateTimeRenderOption", "SERIAL_NUMBER");
  ranges.forEach((range) => url.searchParams.append("ranges", range));
  const response = await fetch(url, { headers: { authorization: `Bearer ${token}` } });
  if (!response.ok) {
    let detail = "";
    try { detail = (await response.json())?.error?.message || ""; } catch {}
    if ((response.status === 429 || response.status >= 500) && attempt < 3) {
      const retryAfter = Number(response.headers.get("retry-after") || 0);
      const waitMs = retryAfter > 0 ? retryAfter * 1000 : 900 * Math.pow(2, attempt);
      await new Promise(resolve => setTimeout(resolve, waitMs));
      return readSheetBatch(token, spreadsheetId, ranges, attempt + 1);
    }
    throw new Error(`UEP 구글시트 읽기 실패 (${response.status})${detail ? `: ${detail}` : ""}`);

  const assigned=new Set();
  (Array.isArray(valueRanges)?valueRanges:[]).forEach((item,index)=>{

    // Sheets batchGet may omit empty ranges or return them in a different order.
    // A response that names its range must only be assigned to that sheet; index is
    // retained solely for legacy responses that contain no range metadata at all.
    const logicalName=responseSheet?logicalBySheet.get(responseSheet):entries[index]?.[0];
    if(!logicalName)return;
    target[logicalName]=item?.values||[];

  const label = button.querySelector("span");
  if (googleConnectionError) {
    button.classList.add("bad");
    if (label) label.textContent = "구글시트 오류";
    button.title = googleConnectionError;
    return;
  }
  if (readonlyCache?.students?.length && neisData?.ok && !neisData.offline) {
    button.classList.add("good");
    if (label) label.textContent = "동기화 정상";
    button.title = "UEP 구글시트와 NEIS 자료가 정상 연결되었습니다.";
    return;
  }
  if (readonlyCache?.students?.length) {
    button.classList.add("warn");
    if (label) label.textContent = neisData?.offline ? "NEIS 저장자료" : "NEIS 연결 대기";
    button.title = neisError || "UEP 구글시트는 연결되었고 NEIS 자료를 확인 중입니다.";
    return;
  }
  if (googleConnectionStatus?.ok) {
    button.classList.add("bad");
    if (label) label.textContent = "시트 권한 확인";
    button.title = googleConnectionError || "서비스 계정 이메일의 구글시트 공유 권한을 확인하세요.";
    return;
  }
  button.classList.add("warn");
  if (label) label.textContent = "인증 필요";
  button.title = "교사 이름·이메일 로그인과 기기승인 상태를 확인하세요.";
}

async function loadNeisDashboard(force = false, requestedDate = "") {
  if (!window.schoolBoard?.loadNeisDashboard) {
    neisError = "이 실행환경에서는 NEIS 연결 기능을 사용할 수 없습니다.";
    updateTopSyncStatus();
    return;
  }
  if (neisLoading) return;
  neisLoading = true;
  if (state.activePage === "dashboard" || state.activePage === "timetable" || state.activePage === "settings") render(state.activePage);
  const options = {
    schoolName: state.settings.neis.schoolName,
    officeCode: state.settings.neis.officeCode,
    schoolCode: state.settings.neis.schoolCode,
    grade: state.settings.neis.grade,
    classNo: state.settings.neis.classNo,
    date: requestedDate || dateKey(new Date()),
    useApiKey: Boolean(state.settings.neis.useApiKey),
    comciganSchoolCode: state.settings.neis.comciganSchoolCode || "69092",
    force,
  };
  try {
    const result = await window.schoolBoard.loadNeisDashboard(options);
    if (!result?.ok) throw new Error(result?.reason || "NEIS 자료를 불러오지 못했습니다.");
    neisData = result;
    neisError = result.offline ? result.staleReason || "실시간 연결 실패로 마지막 저장자료를 표시합니다." : "";
    if (result.school) {
      state.settings.neis.schoolName = result.school.schoolName || state.settings.neis.schoolName;
      state.settings.neis.officeCode = result.school.officeCode || state.settings.neis.officeCode;
      state.settings.neis.schoolCode = result.school.schoolCode || state.settings.neis.schoolCode;
      await save();
    }
  } catch (error) {
    neisError = error?.message || String(error);
    if (!neisData?.ok) neisData = null;
  } finally {
    neisLoading = false;
    updateTopSyncStatus();
    if (["dashboard", "timetable", "settings"].includes(state.activePage)) render(state.activePage);
  }
}

function toggleDashboardEditing() {
  state.settings.layoutLocked = !state.settings.layoutLocked;
  save();
  render("dashboard");
  toast(state.settings.layoutLocked ? "대시보드 배치를 저장하고 잠갔습니다." : "위젯을 끌어서 순서를 바꾸고 크기를 조절하세요.");
}
