async function getSheetsToken(credentials) {
  const now = Math.floor(Date.now() / 1000);
  const header = base64url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claims = base64url(JSON.stringify({
    iss: credentials.client_email,
    scope: "https://www.googleapis.com/auth/spreadsheets",
    aud: credentials.token_uri,
    iat: now,
    exp: now + 3600,
  }));
  const unsigned = `${header}.${claims}`;
  const signature = crypto.sign("RSA-SHA256", Buffer.from(unsigned), credentials.private_key).toString("base64url");
  const body = new URLSearchParams({
    grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
    assertion: `${unsigned}.${signature}`,
  });
  const response = await fetch(credentials.token_uri, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body,
  });
  if (!response.ok) throw new Error(`Google 인증 실패 (${response.status})`);
  const payload = await response.json();

}

async function writeStateNow(value) {
  const target = statePath();
  const dir = path.dirname(target);
  await fs.mkdir(dir, { recursive: true });

  // Each save gets its own temporary file. A fixed ".tmp" name can be
  // removed by a concurrent save, producing ENOENT during copy/rename.
  const temp = path.join(dir, `school-board-state.${process.pid}.${Date.now()}.${crypto.randomBytes(4).toString("hex")}.tmp`);
  const payload = JSON.stringify(value, null, 2);
  try {
    await fs.writeFile(temp, payload, "utf8");
    try {
      await fs.rename(temp, target);
    } catch (renameError) {
      // Windows may reject replacement by rename when the destination exists.
      // Fall back to a direct write; this keeps user data available instead
      // of failing the entire state:save IPC call.
      await fs.writeFile(target, payload, "utf8");
      try { await fs.unlink(temp); } catch {}
    }
  } finally {
    try { await fs.unlink(temp); } catch {}
  }
  return { ok: true, savedAt: new Date().toISOString() };
}
function writeState(value) {
  // Serialize renderer saves so rapid UI updates cannot race each other.
  const task = stateWriteChain.then(() => writeStateNow(value));
  stateWriteChain = task.catch(() => {});
  return task;
}

function encodeGoogleUserOAuthRecovery(value){

let stateWriteChain = Promise.resolve();
async function writeStateNow(value) {
  const target = statePath();
  const dir = path.dirname(target);
  await fs.mkdir(dir, { recursive: true });

  // Each save gets its own temporary file. A fixed ".tmp" name can be
  // removed by a concurrent save, producing ENOENT during copy/rename.
  const temp = path.join(dir, `school-board-state.${process.pid}.${Date.now()}.${crypto.randomBytes(4).toString("hex")}.tmp`);
  const payload = JSON.stringify(value, null, 2);
  try {
    await fs.writeFile(temp, payload, "utf8");
    try {
      await fs.rename(temp, target);
    } catch (renameError) {
      // Windows may reject replacement by rename when the destination exists.
      // Fall back to a direct write; this keeps user data available instead
      // of failing the entire state:save IPC call.
      await fs.writeFile(target, payload, "utf8");
      try { await fs.unlink(temp); } catch {}
    }
  } finally {
    try { await fs.unlink(temp); } catch {}
  }
  return { ok: true, savedAt: new Date().toISOString() };
}
function writeState(value) {
  // Serialize renderer saves so rapid UI updates cannot race each other.
  const task = stateWriteChain.then(() => writeStateNow(value));
  stateWriteChain = task.catch(() => {});
  return task;
}

async function readState() {
  try { return JSON.parse(await fs.readFile(statePath(), "utf8")); }
  catch { return null; }
}
let stateWriteChain = Promise.resolve();
async function writeStateNow(value) {
  const target = statePath();
  const dir = path.dirname(target);
  await fs.mkdir(dir, { recursive: true });

  // Each save gets its own temporary file. A fixed ".tmp" name can be
  // removed by a concurrent save, producing ENOENT during copy/rename.
  const temp = path.join(dir, `school-board-state.${process.pid}.${Date.now()}.${crypto.randomBytes(4).toString("hex")}.tmp`);
  const payload = JSON.stringify(value, null, 2);
  try {
    await fs.writeFile(temp, payload, "utf8");
    try {
      await fs.rename(temp, target);
    } catch (renameError) {
      // Windows may reject replacement by rename when the destination exists.
      // Fall back to a direct write; this keeps user data available instead
      // of failing the entire state:save IPC call.
      await fs.writeFile(target, payload, "utf8");
      try { await fs.unlink(temp); } catch {}
    }
  } finally {
    try { await fs.unlink(temp); } catch {}
  }
  return { ok: true, savedAt: new Date().toISOString() };
}
function writeState(value) {
  // Serialize renderer saves so rapid UI updates cannot race each other.
  const task = stateWriteChain.then(() => writeStateNow(value));
  stateWriteChain = task.catch(() => {});
  return task;
}

function createWindow() {
  const win = new BrowserWindow({
    width: 1500, height: 960, minWidth: 1050, minHeight: 700,
    backgroundColor: "#f4f8fd", title: "UNHO Education Platform (UEP)", frame: false,
    icon: path.join(__dirname, "..", "assets", "unho-uep-icon.png"),
    webPreferences: { preload: path.join(__dirname, "preload.cjs"), contextIsolation: true, nodeIntegration: false },
  });
  mainWindow = win;
  win.setMenuBarVisibility(false);
  win.on("close", () => {
    // A hidden Comcigan BrowserWindow used to keep Electron alive after the main
    // UI disappeared. Destroy auxiliary windows before the last visible window closes.
    destroyAuxiliaryWindows("main-window-close");
  });
  win.on("closed", () => {
    mainWindow = null;
    beginCleanShutdown("main-window-closed");
  });
  win.loadFile(path.join(__dirname, "..", "index.html"));
}
