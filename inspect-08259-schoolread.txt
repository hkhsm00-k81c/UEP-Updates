  try { encrypted = await fs.readFile(filePath); } catch (readError) {
    if (path.resolve(filePath) === path.resolve(credentialPath())) {
      try { return await readCredentialRecovery(); } catch {}
    }
    if (path.resolve(filePath) === path.resolve(credentialPath()) && (readError?.code === "ENOENT" || /no such file/i.test(String(readError?.message||"")))) {
      const friendly = new Error("담임 배포 PC의 Google 시트 연결 승인이 아직 준비되지 않았습니다.");
      friendly.code = "UEP_GOOGLE_CREDENTIAL_MISSING";
      friendly.setupRequired = true;
      throw friendly;
    }
    throw readError;
  }
  if (safeStorage.isEncryptionAvailable()) {
    try { return JSON.parse(safeStorage.decryptString(encrypted)); } catch {}
  }
  // 일부 구버전/수동 이관 파일이 평문 JSON인 경우 현재 형식으로 재저장합니다.
  try {
    const legacy = JSON.parse(encrypted.toString("utf8"));
    if (legacy && typeof legacy === "object") { await saveEncrypted(filePath, legacy); return legacy; }
  } catch {}
  // Google 서비스 계정은 safeStorage가 깨져도 동일 PC/사용자용 AES-GCM 복구본으로 자동 복원합니다.
  if (path.resolve(filePath) === path.resolve(credentialPath())) {
    try {
      const recovered = await readCredentialRecovery();
      if (validateServiceAccount(recovered)) {
        try { if (safeStorage.isEncryptionAvailable()) await fs.writeFile(filePath, safeStorage.encryptString(JSON.stringify(recovered))); } catch {}
        return recovered;
      }
    } catch {}
  }
  const backup = `${filePath}.unreadable-${Date.now()}.bak`;
  try { await fs.rename(filePath, backup); } catch {}
  const error = new Error("저장된 보안 인증정보를 복호화할 수 없습니다. 서비스 계정 JSON을 다시 등록해 주세요.");
  error.code = "UEP_SAFE_STORAGE_DECRYPT_FAILED";
  throw error;
}
