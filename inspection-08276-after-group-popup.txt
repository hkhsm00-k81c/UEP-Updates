function programAreaKind(program) {

  const text = `${program?.type || ""} ${program?.recordTitle || ""} ${program?.actualTitle || program?.title || ""}`;
  if (/진로/.test(text)) return "career";
  return "autonomy";
}
function programAreaLabel(program) {
  const kind = programAreaKind(program);
  if (kind === "career") return "진로";
  if (kind === "after") {
    const text = `${program?.afterType || ""} ${program?.type || ""} ${program?.actualTitle || program?.title || ""}`;
    if (/야간심화|심화수업/.test(text)) return "야간심화수업";
    return "방과후";
  }
  return "자율";
}

function programMonthKey(program) {

  if(direct) return direct[1];

  const year=fallback.match(/(20\d{2})/)?.[1]||String(new Date().getFullYear());
  const month=fallback.match(/(?:^|\D)(1[0-2]|0?[1-9])\s*월/)?.[1];
  return month?`${year}-${String(month).padStart(2,"0")}`:"";
}

function programDurationHours(program){

  if(Number.isFinite(direct)&&direct>0)return direct;

  const time=String(program?.time||"");
  const match=time.match(/(\d{1,2}):(\d{2})\s*[~〜-]\s*(\d{1,2}):(\d{2})/);
  if(match){const start=Number(match[1])*60+Number(match[2]);const end=Number(match[3])*60+Number(match[4]);if(end>start)return Math.round((end-start)/6)/10;}

function normalizeTimetableGradeClass(row){
  const directGrade=Number(row?.grade??row?.학년??0);
  const directClass=Number(row?.classNo??row?.class??row?.반??0);
  if(directGrade>=1&&directGrade<=3&&directClass>=1&&directClass<=20)return {grade:directGrade,classNo:directClass};
  const raw=String(row?.classCode??row?.classroom??row?.학급??row?.반코드??row?.['학년반']??'').trim();
  const digits=raw.replace(/\D/g,'');
  if(digits.length===3){
    const grade=Number(digits[0]),classNo=Number(digits.slice(1));
    if(grade>=1&&grade<=3&&classNo>=1&&classNo<=20)return {grade,classNo};
  }
  if(digits.length===4){
    const grade=Number(digits[0]),classNo=Number(digits.slice(1,3));
    if(grade>=1&&grade<=3&&classNo>=1&&classNo<=20)return {grade,classNo};
  }
  return {grade:directGrade||1,classNo:directClass||Number(digits)||0};
}

function dashboardPersonalWeekMarkup(){
  const {week,days}=personalTimetableData();
  const status=schoolPeriodStatus(new Date());
  const todayIndex=Math.max(0,Math.min(4,(today.getDay()+6)%7));
  const maxPeriods=Math.max(7,Array.isArray(week)?week.length:0);
  if(!Array.isArray(week)||!week.some(row=>Array.isArray(row)&&row.some(Boolean))){
    return `<div class="uep-week-timetable-empty"><span>◷</span><b>내 주간시간표 연결 준비</b><small>컴시간 교사시간표가 연결되면 월~금 수업이 이곳에 표시됩니다.</small><button data-page="timetable">시간표 연결 확인</button></div>`;
  }
  const heads=(days?.length?days:["월","화","수","목","금"]).slice(0,5);
  const cells=[];
  for(let period=1;period<=maxPeriods;period+=1){
    cells.push(`<div class="uep-week-period"><b>${period}</b><small>${escapeHtml(periodTimeLabel(period)||"")}</small></div>`);
    for(let d=0;d<5;d+=1){
      const raw=week[period-1]?.[d]||"";
      const [subject,room,flag]=String(raw).split("|");
      const current=d===todayIndex&&status.index===period-1;
      const classColor=timetableClassColorClass(room);
      cells.push(`<button class="uep-week-class personal-order ${raw?"has-class":"is-empty"} ${classColor} ${current?"is-current":""} ${flag==="changed"?"is-changed":""}" data-page="timetable"><b>${escapeHtml(room||"-")}</b><small>${escapeHtml(subject||"")}</small>${current?'<i aria-label="현재 교시"></i>':''}${flag==="changed"?'<em>변동</em>':''}</button>`);
    }
  }
  return `<div class="uep-week-timetable"><div class="uep-week-corner">교시</div>${heads.map((day,i)=>`<div class="uep-week-head ${i===todayIndex?"today":""}"><b>${escapeHtml(String(day).replace(/요일/g,""))}</b></div>`).join("")}${cells.join("")}</div>`;
}

function recordGradeClass(row){
  const studentNo=String(row?.studentNo||row?.number||row?.studentNumber||'').replace(/\D/g,'');
  const grade=String(row?.grade||row?.gradeNo||row?.studentGrade||(studentNo.length>=4?studentNo[0]:'')||'');

  if(!classNo&&studentNo.length>=4) classNo=String(Number(studentNo.slice(1,3))||'');

function resolveDashboardStudent(row){

  const name=String(row?.name||row?.studentName||row?.displayName||'').trim();

  return students.find(student=>{
    const studentIds=[student?.id,student?.studentId,student?.studentID,student?.studentNo,student?.studentNumber].map(value=>String(value||'').trim()).filter(Boolean);
    if(ids.some(id=>studentIds.includes(id)))return true;
    if(name&&name===String(student?.name||student?.studentName||student?.displayName||'').trim()){
      const studentGc=recordGradeClass(student);
      return (!gc.grade||!studentGc.grade||gc.grade===studentGc.grade)&&(!gc.classNo||!studentGc.classNo||String(Number(gc.classNo))===String(Number(studentGc.classNo)));
    }

}

  state.programEvidence = state.programEvidence || {};
  const current = Array.isArray(state.programEvidence[programId]) ? state.programEvidence[programId] : [];
  const known = new Set(current.map(file=>file.path));
  state.programEvidence[programId] = [...current, ...result.files.filter(file=>!known.has(file.path))];

}

  const lines=[`[UEP 프로그램 안내] ${title}`,`운영일: ${program.date||"미정"}`,program.sessionNo?`회차: ${program.sessionNo}회차`:"",program.status?`수업상태: ${program.status}`:"",program.changeType?`일정변경: ${program.changeType}${program.originalDate?` (원예정일 ${program.originalDate})`:""}`:"",`시간: ${program.time||"미정"}`,`장소: ${program.place||"미정"}`,program.subject?`교과: ${program.subject}`:"",program.department?`담당부서: ${program.department}`:"",program.teacher?`담당교사: ${program.teacher}`:"",program.memo?`안내: ${program.memo}`:"",program.reportFormUrl?`보고서 작성: ${program.reportFormUrl}`:""].filter(Boolean);
  return lines.join("\n");
}

function sendProgramEmailNotice(program){
  const recipients=[...new Set(programStudentMasters(program).map(s=>String(s.email||"").trim()).filter(v=>v&&v!=="-"&&/@/.test(v)))];
  if(!recipients.length)return toast("학생 이메일이 연결된 출석부 인원가 없습니다.");
  const subject=encodeURIComponent(`[UEP] ${program.actualTitle||program.title||"프로그램"} 안내`);
  const body=encodeURIComponent(programNoticeText(program));
  window.open(`mailto:?bcc=${encodeURIComponent(recipients.join(","))}&subject=${subject}&body=${body}`,"_self");
  toast(`${recipients.length}명의 메일 발송 화면을 열었습니다.`);
}

function allConnectedReports(){
  const reports=[];
  (readonlyCache?.students||[]).forEach(student=>(student.reports||[]).forEach(report=>reports.push({...report,__student:student})));
  (readonlyCache?.programs||[]).forEach(program=>(program.reportReports||[]).forEach(report=>reports.push({...report,__program:program})));
  const unique=new Map();
  reports.forEach(report=>{if(report?.id&&!unique.has(String(report.id)))unique.set(String(report.id),report);});
  return [...unique.values()];
}

function parseStudentLine(line) {
  const trimmed = String(line || "").trim();
  if (!trimmed) return { number: "", name: "" };
  const parts = trimmed.includes("\t")
    ? trimmed.split(/\t+/)
    : trimmed.includes(",")
      ? trimmed.split(/,+/)
      : trimmed.split(/\s+/);
  return { number: (parts.shift() || "").trim(), name: parts.join(" ").trim() };
}

    if(sync?.ok&&sync.data){readonlyCache=sync.data;return;}
  }
  const refreshed=await window.schoolBoard?.readReadonlyCache?.();
  if(refreshed?.ok)readonlyCache=refreshed.data;
}
