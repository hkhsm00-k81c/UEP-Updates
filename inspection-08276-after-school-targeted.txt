    .filter((item) => item && typeof item === "object" && String(item.title || "").trim() !== "방학식 학생 안내 확인")
    .map((item) => ({
      ...item,
      requiredClasses: Array.isArray(item.requiredClasses) ? item.requiredClasses : [],
      confirmedClasses: Array.isArray(item.confirmedClasses) ? item.confirmedClasses : [],
    }));

  state.schoolDeadlines = (Array.isArray(state.schoolDeadlines) ? state.schoolDeadlines : structuredClone(baseState.schoolDeadlines)).filter(Boolean);

  state.settings.dashboardOrder = Array.isArray(state.settings.dashboardOrder) ? state.settings.dashboardOrder : [...baseState.settings.dashboardOrder];
  const dashboardKeys = [...baseState.settings.dashboardOrder];
  state.settings.dashboardOrder = [...state.settings.dashboardOrder.filter((key) => dashboardKeys.includes(key)), ...dashboardKeys.filter((key) => !state.settings.dashboardOrder.includes(key))];
  state.settings.neis = { ...baseState.settings.neis, ...(state.settings.neis && typeof state.settings.neis === "object" ? state.settings.neis : {}) };
  state.settings.referenceData = {
    ...baseState.settings.referenceData,
    ...(state.settings.referenceData && typeof state.settings.referenceData === "object" ? state.settings.referenceData : {}),
  };
  for (const key of Object.keys(baseState.settings.referenceData)) {
    if (!Array.isArray(state.settings.referenceData[key])) state.settings.referenceData[key] = structuredClone(baseState.settings.referenceData[key]);
  }
  state.settings.widgetSettings = {
    ...structuredClone(baseState.settings.widgetSettings),
    ...(state.settings.widgetSettings && typeof state.settings.widgetSettings === "object" ? state.settings.widgetSettings : {}),
  };
  for (const [key, fallback] of Object.entries(baseState.settings.widgetSettings)) {
    const current = state.settings.widgetSettings[key];
    state.settings.widgetSettings[key] = current && typeof current === "object" ? { ...fallback, ...current } : structuredClone(fallback);
  }
}

async function withStartupTimeout(promise, milliseconds, fallback = null) {
  let timer;
  try {
    return await Promise.race([
      Promise.resolve(promise),
      new Promise((resolve) => { timer = setTimeout(() => resolve(fallback), milliseconds); }),
    ]);
  } finally {
    if (timer) clearTimeout(timer);
  }
}
