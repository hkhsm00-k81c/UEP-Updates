function calendarEventTargetsFirstGrade(event) {
  const target = String(event?.target || "").replace(/\s+/g, " ").trim();
  const detail = String(event?.detail || "").replace(/\s+/g, " ").replace(/\(?1\s*[,·ㆍ&/]?\s*2?\s*학년부[^)]*\)?/g, " ");
  if (/학부모/.test(target) && !/학생/.test(target)) return false;
  if (/교사|교직원/.test(target) && !/학생/.test(target)) return false;
  const text = target || detail;
  return /(?:^|[^0-9])1\s*학년/.test(text)
    || /1\s*[,·ㆍ&/]\s*2\s*학년/.test(text)
    || /1\s*[~-]\s*3\s*학년/.test(text)
    || /전교생|전체\s*학생/.test(text);
}

}
