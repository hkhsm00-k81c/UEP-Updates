async function getSheetsToken(credentials) {
  const now = Math.floor(Date.now() / 1000);
  const header = base64url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claims = base64url(JSON.stringify({
    iss: credentials.client_email,
    scope: "https://www.googleapis.com/auth/spreadsheets",
    aud: credentials.token_uri,
    iat: now,
    exp: now + 3600,
  }));
  const unsigned = `${header}.${claims}`;
  const signature = crypto.sign("RSA-SHA256", Buffer.from(unsigned), credentials.private_key).toString("base64url");
  const body = new URLSearchParams({
    grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
    assertion: `${unsigned}.${signature}`,
  });
  const response = await fetch(credentials.token_uri, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body,
  });
  if (!response.ok) throw new Error(`Google 인증 실패 (${response.status})`);
  const payload = await response.json();
  if (!payload.access_token) throw new Error("Google 인증 토큰을 받지 못했습니다.");
  return payload.access_token;
}

async function readSheetBatch(token, spreadsheetId, ranges, attempt = 0) {
  if(String(token||'').startsWith('UEP_SCHOOL_READ:')) return schoolReadBatchRead(spreadsheetId,ranges);
  const url = new URL(`https://sheets.googleapis.com/v4/spreadsheets/${spreadsheetId}/values:batchGet`);
  url.searchParams.set("majorDimension", "ROWS");
  url.searchParams.set("valueRenderOption", "UNFORMATTED_VALUE");
  url.searchParams.set("dateTimeRenderOption", "SERIAL_NUMBER");
  ranges.forEach((range) => url.searchParams.append("ranges", range));
  const response = await fetch(url, { headers: { authorization: `Bearer ${token}` } });
  if (!response.ok) {
    let detail = "";
    try { detail = (await response.json())?.error?.message || ""; } catch {}
    if ((response.status === 429 || response.status >= 500) && attempt < 3) {
      const retryAfter = Number(response.headers.get("retry-after") || 0);
      const waitMs = retryAfter > 0 ? retryAfter * 1000 : 900 * Math.pow(2, attempt);
      await new Promise(resolve => setTimeout(resolve, waitMs));
      return readSheetBatch(token, spreadsheetId, ranges, attempt + 1);
    }
    throw new Error(`UEP 구글시트 읽기 실패 (${response.status})${detail ? `: ${detail}` : ""}`);
  }
  return (await response.json()).valueRanges || [];
}

function sheetNameFromA1Range(range="") {
  const prefix=String(range||"").split("!")[0].trim();
  return prefix.replace(/^'/,"").replace(/'$/,"").replace(/''/g,"'");
}

function assignValueRangesByRange(target, entries, valueRanges) {
  const logicalBySheet=new Map(entries.map(([logicalName,range])=>[sheetNameFromA1Range(range),logicalName]));
  const assigned=new Set();
  (Array.isArray(valueRanges)?valueRanges:[]).forEach((item,index)=>{
    const responseSheet=sheetNameFromA1Range(item?.range);
    // Sheets batchGet may omit empty ranges or return them in a different order.
    // A response that names its range must only be assigned to that sheet; index is
    // retained solely for legacy responses that contain no range metadata at all.
    const logicalName=responseSheet?logicalBySheet.get(responseSheet):entries[index]?.[0];
    if(!logicalName)return;
    target[logicalName]=item?.values||[];
    assigned.add(logicalName);
  });
  entries.forEach(([logicalName])=>{if(!assigned.has(logicalName)&&!target[logicalName])target[logicalName]=[];});
  return target;
}

    body: JSON.stringify({ majorDimension: "ROWS", values }),
  });
  if (!response.ok) {
    let detail=""; try{ detail=(await response.json())?.error?.message||""; }catch{}
    if ((response.status===429 || response.status>=500) && attempt<4) {
      const retryAfter=Number(response.headers.get("retry-after")||0);

async function mapWithConcurrency(items, limit, worker) {
  const queue = Array.from(items || []);
  const results = new Array(queue.length);
  let cursor = 0;
  const runners = Array.from({ length: Math.max(1, Math.min(limit, queue.length || 1)) }, async () => {
    while (true) {
      const index = cursor++;
      if (index >= queue.length) return;
      results[index] = await worker(queue[index], index);
    }
  });
  await Promise.all(runners);
  return results;
}
