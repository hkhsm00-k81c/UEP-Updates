function validateServiceAccount(value) {
  return Boolean(
    value && value.type === "service_account" &&
    typeof value.client_email === "string" &&
    typeof value.private_key === "string" &&
    value.private_key.startsWith("-----BEGIN PRIVATE KEY-----") &&
    value.token_uri === "https://oauth2.googleapis.com/token"
  );
}

function recoveryKey(salt) {
  // safeStorage가 Windows/Chromium 키 문제로 복호화에 실패할 때만 사용하는 로컬 복구키입니다.
  // 동일 Windows 사용자/PC에서 재부팅 후에도 안정적으로 재생성됩니다.
  const identity = `${os.hostname()}|${os.userInfo().username}|UEP-CREDENTIAL-RECOVERY-v1`;
  return crypto.scryptSync(identity, salt, 32);
}

}

}

async function mapWithConcurrency(items, limit, worker) {
  const queue = Array.from(items || []);
  const results = new Array(queue.length);
  let cursor = 0;
  const runners = Array.from({ length: Math.max(1, Math.min(limit, queue.length || 1)) }, async () => {
    while (true) {
      const index = cursor++;
      if (index >= queue.length) return;
      results[index] = await worker(queue[index], index);
    }
  });
  await Promise.all(runners);
  return results;
}

function neisRows(payload, key) {
  const group = payload?.[key];
  if (!Array.isArray(group)) return [];
  const rowBlock = group.find((item) => Array.isArray(item?.row));
  return rowBlock?.row || [];
}

function neisResultMessage(payload, key) {

  if (!result) return "";

}

function neisUrl(dataset, params, apiKey = "") {
  const url = new URL(`https://open.neis.go.kr/hub/${dataset}`);
  url.searchParams.set("Type", "json");
  url.searchParams.set("pIndex", "1");
  url.searchParams.set("pSize", "100");
  if (apiKey) url.searchParams.set("KEY", apiKey);
  Object.entries(params || {}).forEach(([key, value]) => {
    if (value !== "" && value !== null && value !== undefined) url.searchParams.set(key, String(value));
  });
  return url.toString();
}

async function getSheetsToken(credentials) {
  const now = Math.floor(Date.now() / 1000);
  const header = base64url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claims = base64url(JSON.stringify({
    iss: credentials.client_email,
    scope: "https://www.googleapis.com/auth/spreadsheets",
    aud: credentials.token_uri,
    iat: now,
    exp: now + 3600,
  }));
  const unsigned = `${header}.${claims}`;
  const signature = crypto.sign("RSA-SHA256", Buffer.from(unsigned), credentials.private_key).toString("base64url");
  const body = new URLSearchParams({
    grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
    assertion: `${unsigned}.${signature}`,
  });
  const response = await fetch(credentials.token_uri, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body,
  });
  if (!response.ok) throw new Error(`Google 인증 실패 (${response.status})`);
  const payload = await response.json();
  if (!payload.access_token) throw new Error("Google 인증 토큰을 받지 못했습니다.");
  return payload.access_token;
}

  if(String(token||'').startsWith('UEP_SCHOOL_READ:')) return schoolReadBatchRead(spreadsheetId,ranges);

    let detail = "";

    if ((response.status === 429 || response.status >= 500) && attempt < 3) {
      const retryAfter = Number(response.headers.get("retry-after") || 0);

  }
  return (await response.json()).valueRanges || [];
}

async function mapWithConcurrency(items, limit, worker) {
  const queue = Array.from(items || []);
  const results = new Array(queue.length);
  let cursor = 0;
  const runners = Array.from({ length: Math.max(1, Math.min(limit, queue.length || 1)) }, async () => {
    while (true) {
      const index = cursor++;
      if (index >= queue.length) return;
      results[index] = await worker(queue[index], index);
    }
  });
  await Promise.all(runners);
  return results;
}

async function getSheetsToken(credentials) {
  const now = Math.floor(Date.now() / 1000);
  const header = base64url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claims = base64url(JSON.stringify({
    iss: credentials.client_email,
    scope: "https://www.googleapis.com/auth/spreadsheets",
    aud: credentials.token_uri,
    iat: now,
    exp: now + 3600,
  }));
  const unsigned = `${header}.${claims}`;
  const signature = crypto.sign("RSA-SHA256", Buffer.from(unsigned), credentials.private_key).toString("base64url");
  const body = new URLSearchParams({
    grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
    assertion: `${unsigned}.${signature}`,
  });
  const response = await fetch(credentials.token_uri, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body,
  });
  if (!response.ok) throw new Error(`Google 인증 실패 (${response.status})`);
  const payload = await response.json();
  if (!payload.access_token) throw new Error("Google 인증 토큰을 받지 못했습니다.");
  return payload.access_token;
}

async function mapWithConcurrency(items, limit, worker) {
  const queue = Array.from(items || []);
  const results = new Array(queue.length);
  let cursor = 0;
  const runners = Array.from({ length: Math.max(1, Math.min(limit, queue.length || 1)) }, async () => {
    while (true) {
      const index = cursor++;
      if (index >= queue.length) return;
      results[index] = await worker(queue[index], index);
    }
  });
  await Promise.all(runners);
  return results;
}

async function getSheetsToken(credentials) {
  const now = Math.floor(Date.now() / 1000);
  const header = base64url(JSON.stringify({ alg: "RS256", typ: "JWT" }));
  const claims = base64url(JSON.stringify({
    iss: credentials.client_email,
    scope: "https://www.googleapis.com/auth/spreadsheets",
    aud: credentials.token_uri,
    iat: now,
    exp: now + 3600,
  }));
  const unsigned = `${header}.${claims}`;
  const signature = crypto.sign("RSA-SHA256", Buffer.from(unsigned), credentials.private_key).toString("base64url");
  const body = new URLSearchParams({
    grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
    assertion: `${unsigned}.${signature}`,
  });
  const response = await fetch(credentials.token_uri, {
    method: "POST",
    headers: { "content-type": "application/x-www-form-urlencoded" },
    body,
  });
  if (!response.ok) throw new Error(`Google 인증 실패 (${response.status})`);
  const payload = await response.json();
  if (!payload.access_token) throw new Error("Google 인증 토큰을 받지 못했습니다.");
  return payload.access_token;
}
