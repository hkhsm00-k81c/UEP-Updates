68: function detectHeaderRowIndex(rows) {
69:   const source = Array.isArray(rows) ? rows : [];
70:   const scanLimit = Math.min(source.length, 12);
71:   let bestIndex = -1;
72:   let bestScore = -Infinity;
73: 
74:   for (let index = 0; index < scanLimit; index += 1) {
75:     const cells = (source[index] || []).map(normalizeHeaderText);
76:     const nonEmpty = cells.filter(Boolean);
77:     if (nonEmpty.length < 2) continue;
78: 
79:     const uniqueCount = new Set(nonEmpty).size;
80:     const duplicatePenalty = nonEmpty.length - uniqueCount;
81:     const knownHeaderHits = nonEmpty.filter((value) =>
82:       /^(학생ID|학번|성명|사용성명|학년|반|번호|일자|운영일|일정ID|프로그램ID|강좌ID|기록ID|상태|구분|설정키|파일구분|스프레드시트ID|탭명|탭용도)$/.test(value),
83:     ).length;
84:     const titlePenalty = nonEmpty.length <= 2 && /UEP|표준|시트|DB|관리|기준/.test(nonEmpty.join(" ")) ? 8 : 0;
85:     const score = nonEmpty.length * 3 + uniqueCount + knownHeaderHits * 8 - duplicatePenalty * 2 - titlePenalty;
86: 
87:     if (score > bestScore) {
88:       bestScore = score;
89:       bestIndex = index;
90:     }
91:   }
92: 
93:   return bestIndex;
94: }

247: function localDate(value) {
248:   if (value === "" || value == null) return "";
249:   if (value instanceof Date && !Number.isNaN(value.getTime()))
250:     return `${value.getFullYear()}-${String(value.getMonth() + 1).padStart(2, "0")}-${String(value.getDate()).padStart(2, "0")}`;
251:   const serialDate = serialToDate(value);
252:   if (serialDate) return serialDate;
253:   const text = String(value).trim();
254:   // __UEP_08159_SCHOOL_READ_DATE_CANONICAL__
255:   if(/^\d{4}-\d{2}-\d{2}T/.test(text)){
256:     const instant=new Date(text);
257:     if(!Number.isNaN(instant.getTime())){
258:       const parts=new Intl.DateTimeFormat('en-CA',{timeZone:'Asia/Seoul',year:'numeric',month:'2-digit',day:'2-digit'}).formatToParts(instant);
259:       const p=Object.fromEntries(parts.map(part=>[part.type,part.value]));
260:       if(p.year&&p.month&&p.day)return p.year+'-'+p.month+'-'+p.day;
261:     }
262:   }
263:   const match = text.match(/(\d{4})[.\-/년]\s*(\d{1,2})[.\-/월]\s*(\d{1,2})/);
264:   if (match) return `${match[1]}-${match[2].padStart(2, "0")}-${match[3].padStart(2, "0")}`;
265:   const parsedText = new Date(text);
266:   if (!Number.isNaN(parsedText.getTime()) && /GMT|UTC|^[A-Za-z]{3}/.test(text))
267:     return `${parsedText.getUTCFullYear()}-${String(parsedText.getUTCMonth() + 1).padStart(2, "0")}-${String(parsedText.getUTCDate()).padStart(2, "0")}`;
268:   const compact=text.replace(/\D/g,'');

280:     const totalMinutes = Math.round((value % 1) * 24 * 60);
281:     return `${String(Math.floor(totalMinutes / 60) % 24).padStart(2, "0")}:${String(totalMinutes % 60).padStart(2, "0")}`;
