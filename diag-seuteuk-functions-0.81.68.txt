function runRecordRuleCheck(text,area){
  const forbidden=recordRuleRowsBySource("35_금지표현DB").map(r=>recordRuleField(r,["금지표현","금지어"])).filter(v=>v&&v.length<60);
  const hits=[...new Set(forbidden.filter(word=>text.includes(word)))].slice(0,10),issues=[];
  if(!text.trim())issues.push("작성된 문장이 없습니다.");
  if(/나는|제가|저는/.test(text))issues.push("학생 자기서술형 표현이 포함되어 있습니다.");
  if(/느꼈|생각했|알게 되었/.test(text))issues.push("단순 소감 표현을 관찰·탐구·성장 중심으로 바꾸는 것이 좋습니다.");
  if(hits.length)issues.push(`금지표현 확인: ${hits.join(", ")}`);
  if(text.length>recordAreaLimit(area))issues.push("영역별 글자 수 제한을 초과했습니다.");
  if(/(?:했다|하였다|했습니다|하였습니다|되었다|이었다)\.?\s*$/.test(text))issues.push("최종 문장이 학생 서술형 ‘~했다’로 끝납니다. 교사 관찰형 종결로 수정해야 합니다.");
  const current=currentRecordStudent();
  const similar=savedDraftsForArea(area,current?.id).map(x=>({...x,score:jaccardSimilarity(text,x.text)})).sort((a,b)=>b.score-a.score)[0];
  if(similar?.score>=0.62){
    const other=(readonlyCache?.students||[]).find(s=>s.id===similar.studentId);
    issues.push(`저장된 ${other?.name||"다른 학생"} 문장과 유사도 ${Math.round(similar.score*100)}%입니다. 활동·역할·강조점을 더 개별화하세요.`);
  }
  return issues;
}

function recordRuleRowsBySource(source){
  return (readonlyCache?.recordRuleRows||[]).filter(row=>String(row.__source||"")===source);
}
function recordRuleField(row,names){
  for(const name of names){
    const key=Object.keys(row||{}).find(k=>String(k).replace(/\s/g,"").includes(name));
    const value=key?String(row[key]??"").trim():"";
    if(value)return value;
  }
  return "";
}
function applySchoolRecordGuardrails(text){
  let out=String(text||"").replace(/\s+/g," ").trim();
  // 33_생기부작성절대규칙과 35_금지표현DB는 학교 검증 규칙이므로 완화하지 않고 최종 강제 적용한다.
  recordRuleRowsBySource("35_금지표현DB").forEach(row=>{
    const forbidden=recordRuleField(row,["금지표현","금지어"]);
    if(!forbidden||forbidden.length>60)return;
    const enabled=recordRuleField(row,["사용여부","활성화여부"]);
    if(enabled&&/^N$/i.test(enabled))return;
    if(!out.includes(forbidden))return;
    const replacement=recordRuleField(row,["수정추천표현","추천표현","대체표현"]);
    const auto=recordRuleField(row,["자동치환여부","자동치환"]);
    if(replacement&&(!auto||/^Y$/i.test(auto)))out=out.split(forbidden).join(replacement);
  });
  return enforceTeacherEnding(out);
}
function recordAdmissionSignal(student){
  const result=[];
  try{
    for(const [group,label] of [["inseoul","인서울"],["regional","지거국"]]){
      const row=admissionManagementDataset(group).find(x=>String(x.student?.id)===String(student?.id));
      if(row&&["core","growth","interest"].includes(row.stage))result.push({group,label,stage:row.stage,reasons:row.reasons||[],managePoints:row.managePoints||[]});
    }
  }catch(e){}
  return result;
}
function recordStudentInterpretation(student,bundle){
  const allRows=recordEvidenceRows(student,bundle,"behavior");
  const reportRows=allRows.filter(r=>r.report);
  const tagText=reportRows.map(r=>recordEvidenceText(r)).join(" ");
  const sdgs=suggestSdgsForRows(reportRows).slice(0,3);
  const careers=CAREER_TAG_CATALOG.map(([label,key,rx])=>({label,key,score:tagMatches(tagText,rx).length})).filter(x=>x.score).sort((a,b)=>b.score-a.score).slice(0,2);
  const majors=MAJOR_TAG_CATALOG.map(([label,rx])=>({label,score:tagMatches(tagText,rx).length})).filter(x=>x.score).sort((a,b)=>b.score-a.score).slice(0,2);
  const activities=bundle.activities||[];
  const insideService=activities.filter(a=>/교육과정\s*내|창체\s*봉사|교내봉사/.test([a.area,a.type,a.title,a.role].filter(Boolean).join(" "))&&!/교육과정\s*외/.test([a.area,a.type,a.title,a.role].filter(Boolean).join(" ")));
  const leadership=activities.filter(a=>/임원|학생회|회장|부회장|실장|부실장|반장|부반장/.test([a.area,a.type,a.title,a.role].filter(Boolean).join(" ")));
  const career=studentCareerSupport(student,bundle);
  const signals=recordAdmissionSignal(student);
  const reportCount=reportRows.length, programCount=(bundle.programs||[]).filter(p=>p.kind!=="after").length;
  const evidenceLevel=reportCount>=3&&programCount>=3?"충분":reportCount||programCount>=2?"보통":"부족";
  const flow=[];
  if(insideService.length)flow.push("교육과정내 봉사");
  if(leadership.length)flow.push("공동체 역할");
  if(programCount)flow.push("프로그램 참여");

  if(careers.length||career.hopeMajor!=="-")flow.push("진로·전공 연결");
  return {sdgs,careers,majors,insideService,leadership,career,signals,reportCount,programCount,evidenceLevel,flow};
}

function buildIntegratedRecordDraft(student,bundle,rows,area,limit){
  let result=buildMultiRecordDraft(rows,area,limit);
  const interp=recordStudentInterpretation(student,bundle);
  const selectedReports=rows.filter(row=>row.report);
  if(selectedReports.length>=2){
    const focus=interp.majors?.[0]?.label||interp.careers?.[0]?.label||interp.sdgs?.[0]?.name||"관련 주제";
    const continuity=`여러 활동에서 ${focus}에 대한 관심을 이어가며 탐구와 성찰의 내용을 연결하는 모습을 보임.`;
    if(!result.includes(continuity)&&result.length+continuity.length+1<=limit)result=`${result} ${continuity}`.trim();
  }
  if(area==="behavior"&&rows.length===0){
    const facts=[];
    if(interp.insideService.length)facts.push("교육과정 내 봉사활동에 참여하며 공동체의 일원으로 주어진 활동을 수행함.");
    if(interp.leadership.length)facts.push("학급 및 학교 공동체에서 맡은 역할을 수행하며 구성원과 함께 활동함.");
    result=facts.join(" ");
  }
  const obs=observationText(student,area);if(obs)result=`${result} ${obs}`.trim();
  result=applySchoolRecordGuardrails(result);
  if(result.length>limit)result=applySchoolRecordGuardrails(result.slice(0,Math.max(20,limit-12)).replace(/\s+\S*$/,""));
  return result.slice(0,limit);
}

function stripProgramDatesFromRecordText(value){
  return String(value||"")
    .replace(/\s*[（(]?(?:20\d{2}[.\-/년]\s*)?\d{1,2}[.\-/월]\s*\d{1,2}(?:일)?[）)]?/g," ")
    .replace(/\s{2,}/g," ").replace(/\s+([,.)])/g,"$1").trim();
}

function runRecordRuleCheck(text,area){
  const forbidden=recordRuleRowsBySource("35_금지표현DB").map(r=>recordRuleField(r,["금지표현","금지어"])).filter(v=>v&&v.length<60);
  const hits=[...new Set(forbidden.filter(word=>text.includes(word)))].slice(0,10),issues=[];
  if(!text.trim())issues.push("작성된 문장이 없습니다.");
  if(/나는|제가|저는/.test(text))issues.push("학생 자기서술형 표현이 포함되어 있습니다.");
  if(/느꼈|생각했|알게 되었/.test(text))issues.push("단순 소감 표현을 관찰·탐구·성장 중심으로 바꾸는 것이 좋습니다.");
  if(hits.length)issues.push(`금지표현 확인: ${hits.join(", ")}`);
  if(text.length>recordAreaLimit(area))issues.push("영역별 글자 수 제한을 초과했습니다.");
  if(/(?:했다|하였다|했습니다|하였습니다|되었다|이었다)\.?\s*$/.test(text))issues.push("최종 문장이 학생 서술형 ‘~했다’로 끝납니다. 교사 관찰형 종결로 수정해야 합니다.");
  const current=currentRecordStudent();
  const similar=savedDraftsForArea(area,current?.id).map(x=>({...x,score:jaccardSimilarity(text,x.text)})).sort((a,b)=>b.score-a.score)[0];
  if(similar?.score>=0.62){
    const other=(readonlyCache?.students||[]).find(s=>s.id===similar.studentId);
    issues.push(`저장된 ${other?.name||"다른 학생"} 문장과 유사도 ${Math.round(similar.score*100)}%입니다. 활동·역할·강조점을 더 개별화하세요.`);
  }
